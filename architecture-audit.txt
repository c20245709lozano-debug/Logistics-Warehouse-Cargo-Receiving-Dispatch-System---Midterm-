<?php

namespace App\Http\Controllers;

use App\Http\Requests\StoreShipmentRequest;
use App\Http\Requests\UpdateShipmentRequest;
use App\Models\Shipment;
use App\Services\ShipmentService;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Inertia\Inertia;
use Inertia\Response;

class ShipmentController extends Controller
{
    protected ShipmentService $shipmentService;

    public function __construct(ShipmentService $shipmentService)
    {
        $this->shipmentService = $shipmentService;
    }

    public function index(Request $request): Response
    {
        $search = trim((string) $request->query('search', ''));
        $status = (string) $request->query('status', 'all');
        $arrivalDate = (string) $request->query('arrival_date', '');

        if (! in_array($status, ['all', 'stored', 'dispatched'], true)) {
            $status = 'all';
        }

        if (
            $arrivalDate !== ''
            && ! preg_match('/^\d{4}-\d{2}-\d{2}$/', $arrivalDate)
        ) {
            $arrivalDate = '';
        }

        $shipments = $this->shipmentService->getPaginatedShipments([
            'search' => $search,
            'status' => $status,
            'arrival_date' => $arrivalDate,
        ]);

        return Inertia::render('shipments/index', [
            'shipments' => $shipments,
            'filters' => [
                'search' => $search,
                'status' => $status,
                'arrival_date' => $arrivalDate,
            ],
        ]);
    }

    public function create(): Response
    {
        return Inertia::render(
            'shipments/create',
            $this->shipmentService->getFormDependencies()
        );
    }

    public function store(StoreShipmentRequest $request): RedirectResponse
    {
        $this->shipmentService->create($request->validated());

        return to_route('shipments.index')
            ->with('success', 'Shipment created successfully.');

            'shipment' => $this->shipmentService->getShipmentWithRelations($shipment),
        ]);
    }

    public function edit(Shipment $shipment): Response
    {
        return Inertia::render('shipments/edit', [
            'shipment' => $this->shipmentService->getShipmentWithRelations($shipment),
            ...$this->shipmentService->getFormDependencies(),
        ]);
    }

    public function update(UpdateShipmentRequest $request, Shipment $shipment): RedirectResponse
    {
        $this->shipmentService->update($shipment, $request->validated());

        return to_route('shipments.index')
            ->with('success', 'Shipment updated successfully.');

    public function destroy(Shipment $shipment): RedirectResponse

<?php

namespace App\Services;

use App\Models\Company;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class CompanyService
{
    public function getPaginatedCompanies(): LengthAwarePaginator
    {
        return Company::withCount(['sentShipments', 'receivedShipments'])
            ->orderBy('company_name')
            ->paginate(10);
    }

    public function getCompanyWithShipmentCounts(Company $company): Company
    {
        return $company->loadCount(['sentShipments', 'receivedShipments']);
    }

    public function create(array $data): Company
    {
        return DB::transaction(function () use ($data) {
            return Company::create($this->sanitizeData($data));
        });
    }

    public function update(Company $company, array $data): Company
    {
        return DB::transaction(function () use ($company, $data) {
            $company->update($this->sanitizeData($data));
            return $company->refresh();
        });
    }

    public function delete(Company $company): bool
    {
        return DB::transaction(function () use ($company) {
            if ($company->sentShipments()->exists() ||
                $company->receivedShipments()->exists()) {
                throw ValidationException::withMessages([
                    'company' => 'This company cannot be deleted because it has shipment records.',
                ]);
            }

            return $company->delete();
        });
    }

    private function sanitizeData(array $data): array
    {
        if (array_key_exists('company_name', $data)) {
            $name = strip_tags((string) $data['company_name']);
            $name = preg_replace('/\s+/u', ' ', $name) ?? $name;
            $data['company_name'] = trim($name);

            if ($data['company_name'] === '') {
                throw ValidationException::withMessages([
                    'company_name' => 'The company name is required.',
                ]);
            }
        }

        return $data;
    }
}

<?php

namespace App\Services;

use App\Models\WarehouseBay;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class WarehouseBayService
{
    public function getPaginatedBays(): LengthAwarePaginator
    {
        return WarehouseBay::withCount([
            'shipments as occupied_shipments_count' => function ($query) {
                $query->where(function ($activeQuery) {
                    $activeQuery->whereNull('dispatch_date')
                        ->orWhereDate('dispatch_date', '>', today());
                });
            },
        ])
            ->orderBy('bay_code')
            ->paginate(10);
    }

    public function getBayWithShipments(WarehouseBay $bay): WarehouseBay
    {
        return $bay->load([
            'shipments.sender',
            'shipments.receiver',
            'shipments.cargoCategory',
        ]);
    }

    public function create(array $data): WarehouseBay
    {
        return WarehouseBay::create($this->sanitizeData($data));
    }

    public function update(
        WarehouseBay $bay,
        array $data
    ): WarehouseBay {
        $data = $this->sanitizeData($data);

        return DB::transaction(function () use ($bay, $data) {
            $bay = WarehouseBay::whereKey($bay->id)
                ->lockForUpdate()
                ->firstOrFail();

            $activeShipments = $bay->shipments()
                ->where(function ($query) {
                    $query->whereNull('dispatch_date')
                        ->orWhereDate('dispatch_date', '>', today());
                })
                ->get();

            $currentWeight = (float) $activeShipments->sum('weight_kg');
            $currentVolume = (float) $activeShipments->sum('volume_cbm');

            $newMaxWeight = array_key_exists('max_weight_kg', $data)
                ? (float) $data['max_weight_kg']
                : (float) $bay->max_weight_kg;

            $newMaxVolume = array_key_exists('max_volume_cbm', $data)
                ? (float) $data['max_volume_cbm']
                : (float) $bay->max_volume_cbm;

            if ($currentWeight > $newMaxWeight ||
                $currentVolume > $newMaxVolume) {
                throw ValidationException::withMessages([
                    'capacity' => 'The new bay capacity cannot be lower than its currently assigned cargo.',
                ]);
            }

            $bay->update($data);

            return $bay->refresh();
        });
    }

    public function delete(WarehouseBay $bay): bool
    {
        return DB::transaction(function () use ($bay) {
            $bay = WarehouseBay::whereKey($bay->id)
                ->lockForUpdate()
                ->firstOrFail();

            if ($bay->shipments()->exists()) {
                return false;
            }

            return $bay->delete();
        });
    }

    private function sanitizeData(array $data): array
    {
        if (array_key_exists('bay_code', $data)) {
            $code = strip_tags((string) $data['bay_code']);
            $code = preg_replace('/\s+/u', ' ', $code) ?? $code;

            // Exam requirement: bay codes must be uppercase.
            $data['bay_code'] = strtoupper(trim($code));

            if ($data['bay_code'] === '') {
                throw ValidationException::withMessages([
                    'bay_code' => 'The bay code is required.',
                ]);
            }
        }

        foreach (['max_weight_kg', 'max_volume_cbm'] as $field) {
            if (array_key_exists($field, $data)) {
                if (! is_numeric($data[$field])) {
                    throw ValidationException::withMessages([
                        $field => 'This value must be numeric.',
                    ]);
                }

                $data[$field] = number_format(
                    (float) $data[$field],
                    2,
                    '.',
                    ''
                );
            }
        }

        return $data;
    }
}

<?php

namespace App\Services;

use App\Models\BillingStatement;
use App\Models\Shipment;
use App\Models\WarehouseBay;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;
use Carbon\Carbon;

 @see \App\Models\Shipment

class ShipmentService
{
    public function create(array $data): Shipment
    {
        return $this->saveShipment(new Shipment(), $data);
    }

    public function update(Shipment $shipment, array $data): Shipment
    {
        return $this->saveShipment($shipment, $data);
    }

    public function delete(Shipment $shipment): bool
    {
        return DB::transaction(function () use ($shipment) {
            $shipment = Shipment::whereKey($shipment->id)
                ->lockForUpdate()
                ->firstOrFail();

            $shipment->billingStatement()->delete();

            return $shipment->delete();
        });
    }

    private function saveShipment(
        Shipment $shipment,
        array $data
    ): Shipment {
        $data = $this->sanitizeData($data);

        return DB::transaction(function () use ($shipment, $data) {
            $shippingFee = array_key_exists('shipping_fee', $data)
                ? (float) $data['shipping_fee']
                : (float) ($shipment->billingStatement?->shipping_fee ?? 0);

            unset($data['shipping_fee']);

            $bayId = $data['warehouse_bay_id']
                ?? $shipment->warehouse_bay_id;

            $bay = WarehouseBay::whereKey($bayId)
                ->lockForUpdate()
                ->firstOrFail();

            $activeShipments = Shipment::where('warehouse_bay_id', $bay->id)
                ->where(function ($query) {
                    $query->whereNull('dispatch_date')
                        ->orWhereDate('dispatch_date', '>', today());
                })
                ->when($shipment->exists, function ($query) use ($shipment) {
                    $query->where('id', '!=', $shipment->id);
                })
                ->get();

            $currentWeight = (float) $activeShipments->sum('weight_kg');
            $currentVolume = (float) $activeShipments->sum('volume_cbm');

            $arrivalDate = Carbon::parse(
                $data['arrival_date'] ?? $shipment->arrival_date
            );

            $dispatchDateValue = $data['dispatch_date']
                ?? $shipment->dispatch_date;

            $dispatchDate = $dispatchDateValue
                ? Carbon::parse($dispatchDateValue)
                : null;

            $willOccupyBay = $dispatchDate === null ||
                $dispatchDate->isAfter(today());

            $newWeight = (float) (
                $data['weight_kg'] ?? $shipment->weight_kg ?? 0
            );

            $newVolume = (float) (
                $data['volume_cbm'] ?? $shipment->volume_cbm ?? 0
            );

            if ($willOccupyBay) {
                if (($currentWeight + $newWeight) > (float) $bay->max_weight_kg) {
                    throw ValidationException::withMessages([
                        'weight_kg' => 'The selected warehouse bay does not have enough weight capacity.',
                    ]);
                }

                if (($currentVolume + $newVolume) > (float) $bay->max_volume_cbm) {
                    throw ValidationException::withMessages([
                        'volume_cbm' => 'The selected warehouse bay does not have enough volume capacity.',
                    ]);
                }
            }

            $shipment->fill($data);
            $shipment->save();

            $storageEndDate = $dispatchDate ?? Carbon::today();

            // Do not calculate negative storage duration for future arrivals.
            $storageDays = $arrivalDate->greaterThan($storageEndDate)
                ? 0
                : $arrivalDate->diffInDays($storageEndDate) + 1;

            $dailyRate = (float) $shipment->cargoCategory()
                ->value('daily_storage_rate');

            $storageFee = round($storageDays * $dailyRate, 2);
            $totalCost = round($storageFee + $shippingFee, 2);

            BillingStatement::updateOrCreate(
                ['shipment_id' => $shipment->id],
                [
                    'storage_fee' => number_format($storageFee, 2, '.', ''),
                    'shipping_fee' => number_format($shippingFee, 2, '.', ''),
                    'total_cost' => number_format($totalCost, 2, '.', ''),
                ]
            );

            return $shipment->load([
                'sender',
                'receiver',
                'cargoCategory',
                'warehouseBay',
                'billingStatement',
            ]);
        });
    }

    private function sanitizeData(array $data): array
    {
        foreach ([
            'sender_company_id',
            'receiver_company_id',
            'cargo_category_id',
            'warehouse_bay_id',
        ] as $field) {
            if (array_key_exists($field, $data) && is_numeric($data[$field])) {
                $data[$field] = (int) $data[$field];
            }
        }

        foreach (['weight_kg', 'volume_cbm', 'shipping_fee'] as $field) {
            if (array_key_exists($field, $data)) {
                if (! is_numeric($data[$field])) {
                    throw ValidationException::withMessages([
                        $field => 'This value must be numeric.',
                    ]);
                }

                $data[$field] = number_format(
                    (float) $data[$field],
                    2,
                    '.',
                    ''
                );
            }
        }

        return $data;
    }
    public function getPaginatedShipments(array $filters = [])
{
    $query = Shipment::query()->with([
        'sender',
        'receiver',
        'cargoCategory',
        'warehouseBay',
        'billingStatement',
    ]);

    if (! empty($filters['search'])) {
        $search = $filters['search'];

        $query->where(function ($q) use ($search) {
            if (ctype_digit($search)) {
                $q->orWhere('id', (int) $search);
            }

            $q->orWhereHas('sender', fn ($c) =>
                $c->where('company_name', 'like', "%{$search}%")
            )->orWhereHas('receiver', fn ($c) =>
                $c->where('company_name', 'like', "%{$search}%")
            )->orWhereHas('cargoCategory', fn ($c) =>
                $c->where('category_name', 'like', "%{$search}%")
            );
        });
    }

    if (($filters['status'] ?? 'all') === 'stored') {
        $query->whereNull('dispatch_date');
    } elseif (($filters['status'] ?? 'all') === 'dispatched') {
        $query->whereNotNull('dispatch_date');
    }

    if (! empty($filters['arrival_date'])) {
        $query->whereDate('arrival_date', $filters['arrival_date']);
    }

    return $query->latest()->paginate(10)->withQueryString();
}

========== app\Http\Middleware\HandleInertiaRequests.php ==========
<?php

namespace App\Http\Middleware;

use Illuminate\Foundation\Inspiring;
use Illuminate\Http\Request;
use Inertia\Middleware;

class HandleInertiaRequests extends Middleware
{
    /**
     * The root template that's loaded on the first page visit.
     *
     * @see https://inertiajs.com/server-side-setup#root-template
     *
     * @var string
     */
    protected $rootView = 'app';

    /**
     * Determines the current asset version.
     *
     * @see https://inertiajs.com/asset-versioning
     */
    public function version(Request $request): ?string
    {
        return parent::version($request);
    }

    /**
     * Define the props that are shared by default.
     *
     * @see https://inertiajs.com/shared-data
     *
     * @return array<string, mixed>
     */
    public function share(Request $request): array
    {
        [$message, $author] = str(Inspiring::quotes()->random())->explode('-');

        return array_merge(parent::share($request), [
            ...parent::share($request),
            'name' => config('app.name'),
            'quote' => ['message' => trim($message), 'author' => trim($author)],
            'auth' => [
                'user' => $request->user(),
            ],
        ]);
    }
}
